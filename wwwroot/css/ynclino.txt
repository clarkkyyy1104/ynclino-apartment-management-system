/* Headings used to be held on one line each. On a wide table — Billing carries
   eleven columns — that alone made the table wider than the space it had, and
   the Update and Delete buttons at the end of every row were pushed off the
   edge. A two-word heading may take two lines; the table then fits. */
thead th {
    padding: .8rem .6rem;
    background: #fff; color: var(--ink); font-weight: 700;
    text-align: center;
    border-bottom: 1px solid var(--line);
}

    padding: .7rem .6rem; text-align: center; vertical-align: middle;
    border-top: 1px solid var(--line-soft);
}
tbody tr:hover { background: #f7f8f9; }
/* The eye was styled by its exact label, so the one view that says
   "View in Tenants" instead of "View" missed the rule and came out looking
   like a different control from the same icon in the other six modules. It is
   matched on being an icon now, not on its wording, and it wears the same box
   as the buttons beside it so the row lines up. */
tbody td a[aria-label^="View"] { color: var(--orange); }
/* An icon button is a fixed square, and the empty slot that stands in for one
   is the same square. Left to size themselves they came out 40px and 35px, and
   the five-pixel difference pushed every button on that row out of line. */
tbody td a[aria-label],
tbody td span[data-ghost][data-icon] {
    width: 2.5rem; padding: 0; justify-content: center;
}

/* ── The actions column ──────────────────────────────────────────────────────
   Buttons were centred in their cell, so a row with three of them centred
   differently from a row with one, and the column came out ragged — 65px of
   drift down the Billing list, 40px down Lost & Found. Anchoring them to the
   right edge gives every row the same edge to line up on, however many buttons
   it happens to have. */
tbody td:last-child { white-space: nowrap; }
tbody td:last-child:has(a, button, span) { text-align: right; }
thead th:last-child { text-align: right; }
table [data-user-actions] { text-align: center !important; }

/* One height for everything that sits in a row, so the icons stop floating
   above the buttons beside them: the eye was a bare 14px glyph next to 33px
   buttons, on a different baseline. */
tbody td a:not([data-record-link]),
tbody td button,
tbody td span[title],
tbody td span[data-ghost] { min-height: 33px; }

/* Every action in a row takes the same width, so a row ending in "Archive"
   lines up with a row ending in "Update" — a word being one letter longer was
   enough to shift the eye beside it six pixels sideways, and no two rows
   agreed. A longer label (Reset Password) simply outgrows the slot; it is the
   same on every row anyway. */
tbody td:last-child a:not([aria-label]),
tbody td:last-child button,
tbody td:last-child span[data-ghost]:not([data-icon]) { min-width: 5.5rem; }
tbody td:last-child form[data-wide-icon-action] button {
    width: 5.5rem;
    padding: .3rem .75rem;
}
tbody td a,
tbody td button,
tbody td form,
tbody td span[title] { vertical-align: middle; }
tbody td:last-child > a,
tbody td:last-child > span { display: inline-block; margin: 0 .2rem; }

/* A button that needs a form around it was getting the form's margin AND its
   own, so it sat a few pixels further along than the plain links next to it and
   the column would not line up. The gap belongs to the control, not to the
   wrapper that happens to be needed to post it. */
tbody td:last-child form { display: inline-block; margin: 0; }
tbody td:last-child form[data-wide-icon-action] { margin: 0 .2rem; }
tbody td:last-child form[data-wide-icon-action] button { margin: 0; }
